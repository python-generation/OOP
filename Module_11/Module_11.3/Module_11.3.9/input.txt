# TEST_1:
request = (
    HttpRequestBuilder()
    .set_url("https://example.com")
    .set_method("POST")
    .set_headers({"Content-Type": "application/json"})
    .set_query_params({"page": 1})
    .set_body('{"name": "Timur"}')
    .set_timeout(5000)
    .build()
)

print(request.url)
print(request.method)
print(request.headers)
print(request.query_params)
print(request.body)
print(request.timeout)
